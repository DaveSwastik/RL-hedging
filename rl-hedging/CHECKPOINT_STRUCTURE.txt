# Method 1: Using helper function (RECOMMENDED)
from load_and_eval import load_checkpoint

actor, critic, quantile_critic = load_checkpoint(
    './runs/training_run_01/checkpoint_5000.pth'
)


# Method 2: Manual loading
import torch
from src.agents.models import MlpActor, MlpCritic
from src.agents.cvar_ctitic import ExDrlCritic

actor = MlpActor(obs_dim=6, hidden_dim=128, action_dim=1)
critic = MlpCritic(obs_dim=6, hidden_dim=128)
quantile_critic = ExDrlCritic(state_dim=6, hidden_dim=128)

checkpoint = torch.load('./runs/training_run_01/checkpoint_5000.pth')
actor.load_state_dict(checkpoint['actor'])
critic.load_state_dict(checkpoint['critic'])
quantile_critic.load_state_dict(checkpoint['quantile_critic'])

actor.eval()
critic.eval()
quantile_critic.eval()

# Now ready for inference!


# ============================================================
#  VERIFYING CHECKPOINT CONTENTS
# ============================================================

# Check what's in the checkpoint
checkpoint = torch.load('./runs/training_run_01/checkpoint_5000.pth')
print(checkpoint.keys())  
# Output: dict_keys(['actor', 'critic', 'quantile_critic']) ✅

# Check model sizes
for model_name, state in checkpoint.items():
    total_params = sum(p.numel() for p in state.values())
    print(f"{model_name}: {total_params} parameters")

# A: Make sure src/agents/models.py and cvar_ctitic.py are importable.
